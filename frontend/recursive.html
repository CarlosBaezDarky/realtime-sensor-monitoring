def find_peaks_recursive(data, index=0, peaks=[]):
    if index >= len(data):
        return peaks
    if 1 ≤ index ≤ len(data)-2:
        if data[index] > data[index-1] and data[index] > data[index+1]:
            peaks.append(index)
    return find_peaks_recursive(data, index + 1, peaks)

def detect_trend_recursive(data, start=0, current_trend=[], all_trends=[]):
    if start >= len(data):
        return all_trends
    if not current_trend:
        return detect_trend_recursive(data, start+1, [start], all_trends)
    last = current_trend[-1]
    if data[start] > data[last]:
        return detect_trend_recursive(data, start+1, current_trend+[start], all_trends)
    else:
        return detect_trend_recursive(data, start, [], all_trends+[current_trend])

    anomalies = []

def max_subarray_recursive(data, low, high):
    if low == high: return (low, high, data[low])
    mid = (low + high) // 2
    left = max_subarray_recursive(data, low, mid)
    right = max_subarray_recursive(data, mid+1, high)
    cross = max_crossing_subarray(data, low, mid, high)
    return max(left, right, cross, key=lambda x: x[2])
